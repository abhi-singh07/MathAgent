import sympy as sp

# Define the vectors
a = sp.Matrix([2, 0, 0])
b = sp.Matrix([5/4, sp.sqrt(39)/4, 0])
c = sp.Matrix([5/4, 15/(4*sp.sqrt(39)), 3*sp.sqrt(39)/13])

#
Compute the dot product a · (b × c)

# The volume is the absolute value of the triple product
volume = abs(triple_product)

print(volume)
