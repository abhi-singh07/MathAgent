import sympy

# Define the range of n based on the problem constraints
n_start = 10
n_end = 14

primes = []

for n in range(n_start, n_end + 1):
    square = n ** 2
    candidates = [square +
1, square + 2]

        if 100 <= num <= 200 and sympy.isprime(num):
