def is_palindrome(n):
    s = str(n)
    return s == s[::-1]

non_palindromes = []

for n in range(10, 100):
    if
is_palindrome(n):
        continue
    current = n
    steps = 0
    while steps < 6:
        reversed_num = int(str(current)[::-1])
        current += reversed_num
        steps += 1
        if is_palindrome(current):
            if steps == 6:
                non_palindromes.append(n)
           
break
    else:
        if is_palindrome(current):
            non_palindromes.append(n)

sum_non_palindromes = sum(non_palindromes)
